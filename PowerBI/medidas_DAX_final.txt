-- Tabla Vuelos

-- Tabla Vuelos
Filas causas excluidas = COUNTROWS(FILTER(Vuelos, NOT(Vuelos[CauseValid])))

-- Tabla Vuelos
Cobertura causas = DIVIDE(CALCULATE([Operaciones], Vuelos[CauseValid] = TRUE()), [Operaciones])

-- Tabla Vuelos
OTP soporte = IF([Operaciones] >= [Soporte minimo], [OTP])

-- Tabla Vuelos
Operaciones soporte = IF([Operaciones] >= [Soporte minimo], [Operaciones])

-- Tabla Vuelos
Cancelacion soporte = IF([Operaciones] >= [Soporte minimo], [Tasa cancelacion])

-- Tabla Vuelos
Referencia OTP = IF([Operaciones] > 0, [Meta OTP])

-- Tabla Vuelos
Wilson inferior = VAR n = [Operaciones] RETURN IF(n >= [Soporte minimo] && n > 0, VAR p = [OTP] VAR z = 1.96 VAR den = 1 + z*z/n VAR c = (p + z*z/(2*n))/den VAR h = z*SQRT(p*(1-p)/n + z*z/(4*n*n))/den RETURN c-h)

-- Tabla Vuelos
Wilson superior = VAR n = [Operaciones] RETURN IF(n >= [Soporte minimo] && n > 0, VAR p = [OTP] VAR z = 1.96 VAR den = 1 + z*z/n VAR c = (p + z*z/(2*n))/den VAR h = z*SQRT(p*(1-p)/n + z*z/(4*n*n))/den RETURN c+h)

-- Tabla Vuelos
Operaciones por estado = SWITCH(SELECTEDVALUE(Estados[Estado]), "Puntuales", [Puntuales], "Demora ≥15", [Demorados 15], "Cancelados", SUM(Vuelos[cancel]), "Desviados", SUM(Vuelos[divert]), "Sin clasificar", [Sin clasificar])

-- Tabla Vuelos
Minutos causa = VAR campo = SELECTEDVALUE(Causas[Campo]) RETURN SWITCH(campo, "carrierDelay", CALCULATE(SUM(Vuelos[carrierDelay]), Vuelos[CauseValid]=TRUE()), "weatherDelay", CALCULATE(SUM(Vuelos[weatherDelay]), Vuelos[CauseValid]=TRUE()), "nasDelay", CALCULATE(SUM(Vuelos[nasDelay]), Vuelos[CauseValid]=TRUE()), "secDelay", CALCULATE(SUM(Vuelos[secDelay]), Vuelos[CauseValid]=TRUE()), "lateAirDelay", CALCULATE(SUM(Vuelos[lateAirDelay]), Vuelos[CauseValid]=TRUE()))

-- Tabla Vuelos
Total minutos conciliados = SUMX(ALL(Causas), [Minutos causa])
